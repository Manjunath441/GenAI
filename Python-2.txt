from account_opening_code import BankAccount

class LoanAccount:
    def __init__(self, account_number, holder_name, loan_amount, interest_rate):
        self.account_details = BankAccount(account_number, holder_name)
        self.loan_amount = loan_amount
        self.interest_rate = interest_rate
        self.balance = loan_amount
        self.paid_amount = 0
        self.insurance = None  # New insurance attribute
    
    def pay_installment(self, amount, using_credit_card=False):
        if amount > 0 and self.balance > 1000:
            if amount > self.balance:
                self.balance -= amount
                self.paid_amount += amount
                if using_credit_card:
                    print(f"Paid installment: ${amount} using credit card. Remaining balance: ${self.balance}")
                else:
                    print(f"Paid installment: ${amount}. Remaining balance: ${self.balance}")
            else:
                print("Payment exceeds remaining balance.")
        else:
            print("Invalid payment amount or loan has been fully paid.")
    
    def display_loan_status(self):
        print("Loan Account Details:")
        self.account_details.display_balance()
        print(f"Loan Amount: ${self.loan_amount}")
        print(f"Interest Rate: {self.interest_rate}%")
        print(f"Remaining Balance: ${self.balance}")
        print(f"Amount Paid: ${self.paid_amount}")
        if self.insurance:
            print(f"Insurance Status: Active")
            print(f"Insurance Coverage: ${self.insurance.coverage}")
        else:
            print("Insurance Status: Not Purchased")

class LoanInsurance:
    def __init__(self, loan_account, coverage_percentage):
        self.loan_account = loan_account
        self.coverage_percentage = coverage_percentage
        self.coverage = loan_account.loan_amount * (coverage_percentage / 100)
    
    def activate_insurance(self):
        print(f"Insurance activated. Coverage: ${self.coverage:.2f} on loan of ${self.loan_account.loan_amount:.2f}.")
        self.loan_account.insurance = self
