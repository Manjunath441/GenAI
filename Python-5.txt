20  def bank_account_services():
21      account_number = input("Enter account number: ")
22      holder_name = input("Enter account holder's name: ")
23      initial_balance = float(input("Enter initial balance (if any): "))
24      bank_account = BankAccount(account_number, holder_name, initial_balance)
25      
26      while True:
27          print("\nSelect an option:")
28          print("1. Deposit")
29          print("2. Withdraw")
30          print("3. Transfer Funds")
31          print("4. Display Balance")
32          print("5. Exit")
33          choice = input("Enter your choice: ")
34          
35          if choice == "1":
36              amount = float(input("Enter deposit amount: "))
37              bank_account.deposit(amount)
38          elif choice == "2":
39              amount = float(input("Enter withdrawal amount: "))
40              bank_account.withdraw(amount)
41          elif choice == "3":
42              target_account_number = input("Enter target account number: ")
43              target_account_holder = input("Enter target account holder name: ")
44              target_account = BankAccount(target_account_number, target_account_holder)
45              amount = float(input("Enter transfer amount: "))
46              bank_account.transfer(target_account, amount)
47          elif choice == "4":
48              bank_account.display_balance()
49          elif choice == "5":
50              print("Exiting...")
51              break

53              print("Invalid choice. Please try again.")
