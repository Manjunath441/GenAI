import time

class User:
    def __init__(self, username, password):
        self.username = username
        self.password = password

class BankAccount:
    def __init__(self, account_number, holder_name, balance=0):
        self.account_number = account_number
        self.holder_name = holder_name
        self.balance = balance
    
    def deposit(self, amount):
        if amount > 0:
            self.balance += amount
            print(f"Deposited ${amount}. Current balance: ${self.balance}")
        else:
            print("Invalid deposit amount.")
    
    def withdraw(self, amount):
        if 0 < amount <= self.balance:
            self.balance -= amount
            print(f"Withdrew ${amount}. Current balance: ${self.balance}")
        else:
            print("Insufficient funds or invalid withdrawal amount.")
    
    def display_balance(self):
        print(f"Account Number: {self.account_number}")
        print(f"Holder Name: {self.holder_name}")
        print(f"Current Balance: ${self.balance}")
    
    def authenticate(self, user):
        username_input = input("Enter username: ")
        password_input = input("Enter password: ")
        if username_input == user.username and password_input == user.password:
            return True
        else:
            print("Invalid credentials. Try again.")
            return False

    def transfer_funds(self, target_account, amount):
        if 0 < amount <= self.balance:
            self.balance -= amount
            target_account.balance += amount
            print(f"Transferred ${amount} to account {target_account.account_number}.")
            print(f"New balance: ${self.balance}")
        else:
            print("Insufficient funds or invalid amount.")

class FixedDeposit:
    def __init__(self, amount, interest_rate, months):
        self.amount = amount
        self.interest_rate = interest_rate
        self.months = months

    def calculate_maturity_amount(self):
        # Using compound interest formula: A = P(1 + r/n)^(nt)
        rate_per_month = self.interest_rate / 100 / 12
        maturity_amount = self.amount * (1 + rate_per_month) ** self.months
        return maturity_amount

    def display_maturity_amount(self):
        maturity_amount = self.calculate_maturity_amount()
        print(f"Fixed Deposit Amount: ${self.amount}")
        print(f"Interest Rate: {self.interest_rate}%")
        print(f"Months: {self.months}")
        print(f"Maturity Amount: ${maturity_amount:.2f}")
