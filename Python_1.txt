 8  class BankAccount:
 9      def __init__(self, account_number, holder_name, balance=0):
10          self.account_number = account_number
11          self.holder_name = holder_name
12          self.balance = balance
13      
14      def deposit(self, amount):
15          if amount > 0:
16              self.balance += amount
17              print(f"Deposited ${amount}. Current balance: ${self.balance}")
18          else:
19              print("Invalid deposit amount.")
20      
21      def withdraw(self, amount):
22          if 0 < amount <= self.balance:
23              self.balance -= amount
24              print(f"Withdrew ${amount}. Current balance: ${self.balance}")
25          else:
26              print("Insufficient funds or invalid withdrawal amount.")
27      
28      def display_balance(self):
29          print(f"Account Number: {self.account_number}")
30          print(f"Holder Name: {self.holder_name}")
31          print(f"Current Balance: ${self.balance}")
32      
33      def authenticate(self, user):
34          username_input = input("Enter username: ")
35          password_input = input("Enter password: ")
36          if username_input == user.username and password_input == user.password:
37              return True
38          else:
39              print("Invalid credentials. Try again.")
40              return False

42      def transfer_funds(self, target_account, amount):
43          if 0 < amount <= self.balance:

45              target_account.balance += amount

51  class FixedDeposit:
52      def __init__(self, amount, interest_rate, months):
53          self.amount = amount
54          self.interest_rate = interest_rate
55          self.months = months
56  
57      def calculate_maturity_amount(self):
58          # Using compound interest formula: A = P(1 + r/n)^(nt)
59          rate_per_month = self.interest_rate / 100 / 12
60          maturity_amount = self.amount * (1 + rate_per_month) ** self.months
61          return maturity_amount
62  
63      def display_maturity_amount(self):
64          maturity_amount = self.calculate_maturity_amount()
65          print(f"Fixed Deposit Amount: ${self.amount}")
66          print(f"Interest Rate: {self.interest_rate}%")
67          print(f"Months: {self.months}")
68          print(f"Maturity Amount: ${maturity_amount:.2f}")
