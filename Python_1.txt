 4  class BankAccount:
 5      def __init__(self, account_number, holder_name, balance=0):
 6          self.account_number = account_number
 7          self.holder_name = holder_name
 8          self.balance = balance
 9  
10      def deposit(self, amount):
11          if amount > 0:
12              self.balance += amount
13              print(f"Deposited ${amount}. Current balance: ${self.balance}")
14          else:
15              print("Invalid deposit amount.")
16  
17      def withdraw(self, amount):
18          if 0 < amount <= self.balance:
19              self.balance -= amount
20              print(f"Withdrew ${amount}. Current balance: ${self.balance}")
21          else:
22              print("Insufficient funds or invalid withdrawal amount.")
23  
24      def display_balance(self):
25          print(f"Account Number: {self.account_number}")
26          print(f"Holder Name: {self.holder_name}")
27          print(f"Current Balance: ${self.balance}")
28  
29      def transfer(self, target_account, amount):
30          if 0 < amount <= self.balance:
31              self.balance -= amount
32              target_account.balance += amount
33              print(f"Transferred ${amount} to {target_account.holder_name}. New balance: ${self.balance}")
34          else:
35              print("Insufficient funds or invalid transfer amount.")
